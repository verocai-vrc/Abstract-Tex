\newcommand{\pipelinefigure}[1]{%
  \begin{tikzpicture}[node distance=1.6cm, box/.style={draw, rounded corners, minimum width=2.1cm, minimum height=0.8cm, fill=#1!12}]
    \node[box] (edit) {Editor};
    \node[box, right=of edit] (disk) {File};
    \node[box, right=of disk] (engine) {Engine};
    \node[box, right=of engine] (pdf) {PDF};
    \draw[-Stealth] (edit) -- node[above]{\small 700 ms} (disk);
    \draw[-Stealth] (disk) -- (engine);
    \draw[-Stealth] (engine) -- (pdf);
    \draw[-Stealth, dashed] (pdf.south) to[bend left=20] node[below]{\small SyncTeX} (edit.south);
  \end{tikzpicture}}

    \end{axis}
